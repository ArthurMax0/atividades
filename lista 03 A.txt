import java.util.Scanner;
public class Exercicio4 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int opcao;
        do {
            System.out.println();
            System.out.println("1 - Dobro");
            System.out.println("2 - Metade");
            System.out.println("3 - Quadrado");
            System.out.println("0 - Sair");
            System.out.print("Escolha uma opção: ");
            opcao = sc.nextInt();
            switch (opcao) {
                case 1:
                    System.out.print("Digite um numero: ");
                    double numero1 = sc.nextDouble();
                    System.out.println("Dobro: " + (numero1 * 2));
                    break;
                case 2:
                    System.out.print("Digite um numero: ");
                    double numero2 = sc.nextDouble();
                    System.out.println("Metade: " + (numero2 / 2));
                    break;
                case 3:
                    System.out.print("Digite um numero: ");
                    double numero3 = sc.nextDouble();
                    System.out.println("Quadrado: " + (numero3 * numero3));
                    break;
                case 0:
                    System.out.println("Encerrando");
                    break;
                default:
                    System.out.println("Opcao invalida");
            }
        } while (opcao != 0);
       
 sc.close();
    }
}

import java.util.Scanner;
public class Exercicio5 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Digite A: ");
        int a = sc.nextInt();
        System.out.print("Digite B: ");
        int b = sc.nextInt();
        if (a > b) {
            System.out.println("Intervalo invalido");
        } else {
            int soma = 0;
            for (int i = a; i <= b; i++) {
                if (i % 2 == 0) {
                    soma += i;
                }
            }
            System.out.println("Soma dos numeros pares: " + soma);
        }

        sc.close();
    }
}

import java.util.Scanner;
public class Exercicio7 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Quantidade de alunos: ");
        int quantidade = sc.nextInt();
        double soma = 0;
        int aprovados = 0;
        int abaixoDeSeis = 0;
        for (int i = 1; i <= quantidade; i++) {
            System.out.print("Nota do aluno " + i + ": ");
            double nota = sc.nextDouble();
            soma += nota;
            if (nota >= 6) {
                aprovados++;
            } else {
                abaixoDeSeis++;
            }
        }
        if (quantidade > 0) {
            double media = soma / quantidade;
            System.out.printf("Media da turma: %.2f%n", media);
            System.out.println("Alunos com nota maior ou igual a 6: " + aprovados);
            System.out.println("Alunos com nota abaixo de 6: " + abaixoDeSeis);
        } else {
            System.out.println("Quantidade de alunos sei la.");
        }

        sc.close();
    }
}

import java.util.Scanner;
public class Exercicio8 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int quantidade = 0;
        int soma = 0;
        int maior = 0;
        int menor = 0;
        System.out.print("Digite um valor (0 para encerrar): ");
        int numero = sc.nextInt();
        while (numero != 0) {
            if (quantidade == 0) {
                maior = numero;
                menor = numero;
            }
            soma += numero;
            quantidade++;
            if (numero > maior) {
                maior = numero;
            }
            if (numero < menor) {
                menor = numero;
            }
            System.out.print("Digite outro valor (0 para encerrar): ");
            numero = sc.nextInt();
        }
        if (quantidade == 0) {
            System.out.println("Nenhum valor foi colocado.");
        } else {
            System.out.println("Quantidade: " + quantidade);
            System.out.println("Soma: " + soma);
            System.out.println("Maior: " + maior);
            System.out.println("Menor: " + menor);
        }
        sc.close();
    }
}

import java.util.Scanner;
public class Exercicio10 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int codigo;
        int totalItens = 0;
        double totalCompra = 0;
        System.out.println("CAIXA DO SUPERMERCADO");
        System.out.print("Codigo do produto (0 para finalizar): ");
        codigo = sc.nextInt();
        while (codigo != 0) {
            System.out.print("Preco unitario: R$ ");
            double preco = sc.nextDouble();
            System.out.print("Quantidade: ");
            int quantidade = sc.nextInt();
            totalCompra += preco * quantidade;
            totalItens += quantidade;
            System.out.print("Codigo do produto (0 para finalizar): ");
            codigo = sc.nextInt();
        }
        System.out.printf("Total da compra: R$ %.2f%n", totalCompra);
        System.out.println("Quantidade de itens: " + totalItens);
        System.out.println();
        System.out.println("Forma de pagamento:");
        System.out.println("1 - Dinheiro");
        System.out.println("2 - Debito");
        System.out.println("3 - Credito");
        System.out.print("Opcao: ");
        int pagamento = sc.nextInt();
        double totalFinal = totalCompra;
        switch (pagamento) {
            case 1:
                totalFinal = totalCompra - (totalCompra * 0.05);
                System.out.println("Pagamento em dinheiro.");
                System.out.println("Desconto de 5%.");
                break;
            case 2:
                System.out.println("Pagamento no debito.");
                break;
            case 3:
                totalFinal = totalCompra + (totalCompra * 0.03);
                System.out.println("Pagamento no credito.");
                System.out.println("Acrescimo de 3%.");
                break;
            default:
                System.out.println("Forma de pagamento invalida.");
        }
        if (totalCompra > 300) {
            System.out.println("Aviso: compra grande.");
        }
        if (totalItens > 20) {
            System.out.println("Aviso: muitos itens.");
        }
        System.out.printf("Total final: R$ %.2f%n", totalFinal);
       
 sc.close();
    }
}
